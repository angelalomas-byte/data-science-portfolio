#Creación de la función
resumen_activo <- function(ticker, lista_xts){

  x <- lista_xts[[ticker]]

  # 1) ¿Close o Adjusted?
  pct_neg <- mean(Ad(x) < 0, na.rm = TRUE) * 100
  
  if(pct_neg > 5){
    serie <- Cl(x)
    tipo  <- "Close"
  } else {
    serie <- Ad(x)
    tipo  <- "Adjusted"
  }

  # 2) Fechas de inicio y fin
  F_ini <- as.Date(start(serie))
  F_fin <- as.Date(end(serie))

  # 3) Rendimiento logarítmico
  rend <- diff(log(serie))
  rend <- as.numeric(rend)
  
  # T0 observaciones iniciales
  T0 <- sum(!is.na(rend))

  # 4) Porcentaje de NA
  pct_NA <- mean(is.na(rend)) * 100

  # 5) Outliers con C = 3 y C = 5
  z <- scale(rend)
  out3 <- sum(abs(z) > 3, na.rm = TRUE)
  out5 <- sum(abs(z) > 5, na.rm = TRUE)

  # 6) T1 tras limpiar con C = 5
  rend_clean <- rend[abs(z) <= 5]
  T1 <- sum(!is.na(rend_clean))

  # Tabla final para este activo
  data.frame(
    Ticker   = ticker,
    Serie    = tipo,
    Fecha_Inicio    = F_ini,
    Fecha_Fin    = F_fin,
    Tamaño_Inicial      = T0,
    Porcentaje_NaN   = round(pct_NA, 2),
    C3       = out3,
    C5       = out5,
    Tamaño_Final       = T1,
    stringsAsFactors = FALSE
  )
}

# Función para analizar la HME de cada activo
analizar_activo <- function(precio, rend, nombre_activo = 'Activo') {
  

# ----------------------
# ESTACIONARIEDAD PRECIOS
# ----------------------
precio_clean <- na.omit(precio)  # elimina NAs
adf_val_precio <- tryCatch(adf.test(precio_clean)$p.value, error = function(e) NA)
pp_val_precio  <- tryCatch(pp.test(precio_clean)$p.value,  error = function(e) NA)
kpss_val_precio <- tryCatch(kpss.test(precio_clean)$p.value, error = function(e) NA)

estacionariedad_precio <- if(!is.na(adf_val_precio) & !is.na(pp_val_precio) & !is.na(kpss_val_precio)) {
  if(adf_val_precio < 0.05 & pp_val_precio < 0.05 & kpss_val_precio > 0.05) {
    "Precios estacionarios"
  } else {
    "Precios no estacionarios"
  }
} else {
  "Error test estacionariedad"
}


# ----------------------
# ESTACIONARIEDAD RENDIMIENTOS
# ----------------------
adf_val <- adf.test(rend)$p.value
pp_val  <- pp.test(rend)$p.value
kpss_val <- kpss.test(rend)$p.value

estacionariedad_rend <- if(adf_val < 0.05 & pp_val < 0.05 & kpss_val > 0.05) {
  "Rendimientos estacionarios"
} else {
  "No estacionario"
}


# ----------------------
# INCORRELACIÓN
# ----------------------
ljung <- Box.test(rend, lag = 5, type = "Ljung-Box")$p.value

incorrelacion_rend <- if(ljung > 0.05) {
  "A favor HME"
} else {
  "En contra HME"
}



# ----------------------
# INCORRELACIÓN rendimientos al cuadrado
# ----------------------
ljung2 <- Box.test(rend^2, lag = 5, type = "Ljung-Box")$p.value

incorrelacion_rend2 <- if(ljung2 > 0.05) {
  "A favor HME"
} else {
  "En contra HME"
}

  
# ----------------------
# EXPONENTE DE HURST
# ----------------------
hurst_val <- tryCatch(hurstexp(rend)$Hs, error=function(e) NA)
  hurst <- if(!is.na(hurst_val)) {
    if(hurst_val < 0.5) "RR. No cumple HME"
    else if(hurst_val == 0.5) "RB. Se cumple HME"
    else "RN. No cumple HME"
  } else {
    "Error Hurst"
  }
  
# ----------------------
# TEST DE BDS
# ----------------------
bds_val <- tryCatch(bds.test(rend)$p.value, error = function(e) NA)

bds <- if (any(is.na(bds_val))) {
  "Error BDS"
} else if (all(bds_val > 0.05)) {
  "Se cumple HME"
} else {
  "No cumple HME"
}

  
# ----------------------
# NORMALIDAD
# ----------------------
jb_val <- tryCatch(jarque.bera.test(rend)$p.value, error = function(e) NA)
normalidad <- if (is.na(jb_val)) {
  "Error normalidad"
} else if (jb_val > 0.05) {
  "Normalidad compatible RB"
} else {
  "Distribución no normal"
}


# ----------------------
# Determinar HME resumen
# ----------------------
hme_resumen <- if(
    incorrelacion_rend == "A favor HME" &
    incorrelacion_rend2 == "A favor HME" &
    grepl("RB", hurst) &
    bds == "Se cumple HME"
  ) "Sí" else "No"


# ----------------------
#MODELO
# ----------------------
modelo <- if(incorrelacion_rend == "En contra HME" & incorrelacion_rend2 == "En contra HME") {
    "ARMA-GARCH"
  } else if(incorrelacion_rend == "A favor HME" & incorrelacion_rend2 == "En contra HME") {
    "GARCH"
  } else {
    "No aplica"
  }



# ----------------------
# GRÁFICOS
# ----------------------
subfolder <- file.path(out_dir, nombre_activo)
  if(!dir.exists(subfolder)) dir.create(subfolder)

# ACF / PACF con ggplot
g_acf <- ggAcf(rend) + ggtitle(paste0("ACF - Rendimientos ", nombre_activo))
ggsave(file.path(subfolder, "ACF_rendimientos.png"), g_acf, width=8, height=6)

g_pacf <- ggPacf(rend) + ggtitle(paste0("PACF - Rendimientos ", nombre_activo))
ggsave(file.path(subfolder, "PACF_rendimientos.png"), g_pacf, width=8, height=6)

g_acf2 <- ggAcf(rend^2) + ggtitle(paste0("ACF - Rendimientos^2 ", nombre_activo))
ggsave(file.path(subfolder, "ACF_rend2.png"), g_acf2, width=8, height=6)

g_pacf2 <- ggPacf(rend^2) + ggtitle(paste0("PACF - Rendimientos^2 ", nombre_activo))
ggsave(file.path(subfolder, "PACF_rend2.png"), g_pacf2, width=8, height=6)

  
  
# Serie con bandas ±2SD
serie <- as.numeric((rend))
df <- data.frame(Fecha = 1:length(serie), Rendimiento = serie)
media <- mean(df$Rendimiento)
sd_val <- sd(df$Rendimiento)
upper <- media + 2*sd_val
lower <- media - 2*sd_val

g_band <- ggplot(df, aes(x = Fecha, y = Rendimiento)) +
  geom_line(color = "blue") +
  geom_hline(yintercept = media, linetype = "dashed", color = "black") +
  geom_hline(yintercept = upper, linetype = "dotted", color = "red") +
  geom_hline(yintercept = lower, linetype = "dotted", color = "red") +

  Incorrelacion_Rend2 = incorrelacion_rend2,
  Hurst = hurst,
  BDS = bds,
  Normalidad = normalidad,

  Modelo = modelo
)
}
